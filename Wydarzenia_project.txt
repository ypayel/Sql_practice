-- TABELA: Harmonogram

CREATE TABLE Harmonogram (
    id_harmonogramu INT IDENTITY PRIMARY KEY,
    id_wydarzenia INT,
    id_sali INT,
    data DATE,
    godzina_start TIME,
    godzina_koniec TIME,
    FOREIGN KEY (id_wydarzenia) REFERENCES Wydarzenia(id_wydarzenia),
    FOREIGN KEY (id_sali) REFERENCES Sale(id_sali)
);

-- TABELA: Zapisy

CREATE TABLE Zapisy (
    id_zapisu INT IDENTITY PRIMARY KEY,
    id_uzytkownika INT,
    id_wydarzenia INT,
    data_zapisu DATE DEFAULT GETDATE(),
    status NVARCHAR(20) CHECK (status IN ('aktywny','anulowany')),
    obecny BIT DEFAULT 0,
    FOREIGN KEY (id_uzytkownika) REFERENCES Uzytkownicy(id_uzytkownika),
    FOREIGN KEY (id_wydarzenia) REFERENCES Wydarzenia(id_wydarzenia)
);


-- TABELA: Platnosci

CREATE TABLE Platnosci (
    id_platnosci INT IDENTITY PRIMARY KEY,
    id_zapisu INT UNIQUE,
    kwota DECIMAL(8,2),
    data_platnosci DATE,
    metoda NVARCHAR(30) CHECK (metoda IN ('karta','gotowka','przelew')),
    oplacona BIT DEFAULT 0,
    FOREIGN KEY (id_zapisu) REFERENCES Zapisy(id_zapisu)
);


-- TABELA: Opinie
CREATE TABLE Opinie (
    id_opinii INT IDENTITY PRIMARY KEY,
    id_wydarzenia INT,
    id_uzytkownika INT,
    ocena INT CHECK (ocena BETWEEN 1 AND 5),
    komentarz NVARCHAR(300),
    data_dodania DATE DEFAULT GETDATE(),
    FOREIGN KEY (id_wydarzenia) REFERENCES Wydarzenia(id_wydarzenia),
    FOREIGN KEY (id_uzytkownika) REFERENCES Uzytkownicy(id_uzytkownika)
);


-- TABELA: Uzytkownicy_Role

CREATE TABLE Uzytkownicy_Role (
    id INT IDENTITY PRIMARY KEY,
    id_uzytkownika INT,
    id_roli INT,
    data_nadania DATE DEFAULT GETDATE(),
    aktywna BIT DEFAULT 1,
    uwagi NVARCHAR(200),
    FOREIGN KEY (id_uzytkownika) REFERENCES Uzytkownicy(id_uzytkownika),
    FOREIGN KEY (id_roli) REFERENCES Role(id_roli)
);


-- WIDOKI

CREATE VIEW Widok_Wydarzenia_Zapisy AS
SELECT w.nazwa, COUNT(z.id_zapisu) AS liczba_zapisow
FROM Wydarzenia w
LEFT JOIN Zapisy z ON w.id_wydarzenia = z.id_wydarzenia
GROUP BY w.nazwa;

GO

CREATE VIEW Widok_Uzytkownicy_Role AS
SELECT u.imie, u.nazwisko, r.nazwa_roli
FROM Uzytkownicy u
JOIN Uzytkownicy_Role ur ON u.id_uzytkownika = ur.id_uzytkownika
JOIN Role r ON ur.id_roli = r.id_roli;
GO


-- PROCEDURY

CREATE PROCEDURE DodajZapis
    @id_uzytkownika INT,
    @id_wydarzenia INT
AS
BEGIN
    INSERT INTO Zapisy(id_uzytkownika, id_wydarzenia, status)
    VALUES (@id_uzytkownika, @id_wydarzenia, 'aktywny');
END;
GO

CREATE PROCEDURE ZmienStatusPlatnosci
    @id_zapisu INT
AS
BEGIN
    UPDATE Platnosci
    SET oplacona = 1, data_platnosci = GETDATE()
    WHERE id_zapisu = @id_zapisu;
END;
GO
